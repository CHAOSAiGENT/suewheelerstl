```sql
-- ============================================================
-- MVP2 migration: new pipeline statuses, crew roster, bid blocks
-- ============================================================

-- 1. Expand status on submissions
--    Old values (new, reviewed, quote_sent, scheduled, completed) remain valid
--    during cutover. A separate data migration maps them after deploy.
ALTER TABLE submissions
  ADD COLUMN IF NOT EXISTS bid_start_date          date,
  ADD COLUMN IF NOT EXISTS crew_member_ids         uuid[]      NOT NULL DEFAULT '{}',
  ADD COLUMN IF NOT EXISTS lost_reason             text        CHECK (lost_reason IN ('price','timing','competitor','unknown')),
  ADD COLUMN IF NOT EXISTS lost_note               text,
  ADD COLUMN IF NOT EXISTS after_photos_uploaded   boolean     NOT NULL DEFAULT false,
  ADD COLUMN IF NOT EXISTS final_payment_confirmed boolean     NOT NULL DEFAULT false,
  ADD COLUMN IF NOT EXISTS archived_at             timestamptz;

-- Allow new status values (column is text — no enum to alter)
-- Existing CHECK constraint on status must be dropped and recreated.
-- If no CHECK exists this is a no-op drop; the CREATE adds the constraint.
ALTER TABLE submissions
  DROP CONSTRAINT IF EXISTS submissions_status_check;

ALTER TABLE submissions
  ADD CONSTRAINT submissions_status_check
  CHECK (status IN (
    'new','reviewed','quote_sent','scheduled','completed',
    'to_bid','bid_sent','on_deck','active','finishing','fully_complete','lost'
  ));

-- 2. Crew members table
CREATE TABLE IF NOT EXISTS crew_members (
  id         uuid        PRIMARY KEY DEFAULT gen_random_uuid(),
  name       text        NOT NULL,
  active     boolean     NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now()
);

ALTER TABLE crew_members ENABLE ROW LEVEL SECURITY;
-- Service role bypasses RLS; no client-side access needed.

-- 3. Bid blocks table
CREATE TABLE IF NOT EXISTS bid_blocks (
  id            uuid        PRIMARY KEY DEFAULT gen_random_uuid(),
  submission_id uuid        NOT NULL REFERENCES submissions(id) ON DELETE CASCADE,
  sort_order    int         NOT NULL DEFAULT 0,
  source_key    text        NOT NULL,
  title         text        NOT NULL,
  body          text        NOT NULL,
  photo_paths   text[]      NOT NULL DEFAULT '{}',
  created_at    timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS bid_blocks_submission_idx
  ON bid_blocks(submission_id, sort_order);

ALTER TABLE bid_blocks ENABLE ROW LEVEL SECURITY;
-- Service role only; no public access.

-- 4. Data migration: map old status values to new pipeline
UPDATE submissions SET status = 'to_bid'        WHERE status IN ('new','reviewed');
UPDATE submissions SET status = 'bid_sent'      WHERE status = 'quote_sent';
UPDATE submissions SET status = 'on_deck'       WHERE status = 'scheduled';
UPDATE submissions SET status = 'fully_complete' WHERE status = 'completed';
```

- [ ] **Update the PATCH route to accept new fields**

Replace `src/app/api/admin/submissions/[id]/route.ts`:

```typescript
import { NextResponse } from "next/server";
import { createServerSupabaseClient } from "@/lib/supabase/server";
import { createServiceSupabaseClient } from "@/lib/supabase/service";
import type { SubmissionStatus, LostReason } from "@/lib/types";
import { KANBAN_COLUMNS } from "@/lib/types";

export async function PATCH(
  req: Request,
  { params }: { params: Promise<{ id: string }> },
) {
  const supabaseAuth = await createServerSupabaseClient();
  const {
    data: { user },
  } = await supabaseAuth.auth.getUser();
  if (!user)
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

  const { id } = await params;
  const body = await req.json();
  const update: Record<string, unknown> = {};

  if (body.status !== undefined) {
    if (!KANBAN_COLUMNS.includes(body.status)) {
      return NextResponse.json({ error: "Invalid status" }, { status: 400 });
    }
    update.status = body.status;
  }
  if (body.admin_notes !== undefined) update.admin_notes = body.admin_notes;
  if (body.lost_reason !== undefined) update.lost_reason = body.lost_reason;
  if (body.lost_note !== undefined) update.lost_note = body.lost_note;
  if (body.crew_member_ids !== undefined)
    update.crew_member_ids = body.crew_member_ids;
  if (body.after_photos_uploaded !== undefined)
    update.after_photos_uploaded = body.after_photos_uploaded;
  if (body.final_payment_confirmed !== undefined)
    update.final_payment_confirmed = body.final_payment_confirmed;

  if (Object.keys(update).length === 0) {
    return NextResponse.json({ error: "Nothing to update" }, { status: 400 });
  }

  const supabase = createServiceSupabaseClient();
  const { data, error } = await supabase
    .from("submissions")
    .update(update)
    .eq("id", id)
    .select()
    .single();

  if (error)
    return NextResponse.json({ error: "Update failed" }, { status: 500 });
  return NextResponse.json({ ok: true, submission: data });
}
```

```typescript
"use client";

import { useDraggable } from "@dnd-kit/core";
import { CSS } from "@dnd-kit/utilities";
import type { Submission } from "@/lib/types";
import { STATUS_COLORS } from "@/lib/types";

interface Props {
  submission: Submission;
  crewNames: Record<string, string>; // id → name
}

export function KanbanCard({ submission, crewNames }: Props) {
  const { attributes, listeners, setNodeRef, transform, isDragging } = useDraggable({
    id: submission.id,
    data: { submission },
  });

  const style = {
    transform: CSS.Translate.toString(transform),
    opacity: isDragging ? 0.4 : 1,
    cursor: isDragging ? "grabbing" : "grab",
  };

  const lastName = submission.name.split(" ").pop() ?? submission.name;
  const addressShort = submission.address ?? "No address";
  const crewTags = submission.crew_member_ids
    .map((id) => crewNames[id])
    .filter(Boolean);
  const noteSnippet = submission.admin_notes
    ? submission.admin_notes.slice(0, 60) + (submission.admin_notes.length > 60 ? "…" : "")
    : null;
  const startDate = submission.bid_start_date
    ? new Date(submission.bid_start_date).toLocaleDateString("en-US", { month: "short", day: "numeric" }).toUpperCase()
    : "TBD";
  const colColor = STATUS_COLORS[submission.status];

  return (
    <div
      ref={setNodeRef}
      style={style}
      {...listeners}
      {...attributes}
      className="bg-white border border-[rgba(42,36,33,0.1)] rounded-sm p-3 shadow-sm select-none hover:border-[#11B2E8] transition-colors"
    >
      <a
        href={`/admin/${submission.id}`}

        className="block"
        onPointerDown={(e) => e.stopPropagation()}
```

```typescript
import { useState, useCallback } from "react";
import {
  DndContext,
  DragOverlay,
  PointerSensor,
  KeyboardSensor,
  useSensor,
  useSensors,
  type DragEndEvent,
  type DragStartEvent,
} from "@dnd-kit/core";
import type { Submission, SubmissionStatus, LostReason, CrewMember } from "@/lib/types";
import { KANBAN_COLUMNS, STATUS_LABELS } from "@/lib/types";
import { KanbanColumn } from "./KanbanColumn";
import { KanbanCard } from "./KanbanCard";
import { LostReasonModal } from "./LostReasonModal";
import { FinishingGateModal } from "./FinishingGateModal";

interface Props {
  initialSubmissions: Submission[];
  crewMembers: CrewMember[];
}

export function KanbanBoard({ initialSubmissions, crewMembers }: Props) {
  const [submissions, setSubmissions] = useState(initialSubmissions);
  const [draggingId, setDraggingId] = useState<string | null>(null);

  // Pending move while waiting for modal confirmation
  const [pendingMove, setPendingMove] = useState<{
    id: string;
    from: SubmissionStatus;
    to: SubmissionStatus;
  } | null>(null);

  const crewNames = Object.fromEntries(crewMembers.map((c) => [c.id, c.name]));

  const sensors = useSensors(
    useSensor(PointerSensor, { activationConstraint: { distance: 5 } }),
    useSensor(KeyboardSensor)
  );

  const grouped = KANBAN_COLUMNS.reduce<Record<SubmissionStatus, Submission[]>>(
    (acc, col) => {
      acc[col] = submissions.filter((s) => s.status === col && !s.archived_at);
      return acc;
    },
    {} as Record<SubmissionStatus, Submission[]>
  );

  const draggingSubmission = draggingId
    ? submissions.find((s) => s.id === draggingId) ?? null
    : null;

  function handleDragStart(event: DragStartEvent) {
    setDraggingId(event.active.id as string);
  }

  function handleDragEnd(event: DragEndEvent) {
    setDraggingId(null);
    const { active, over } = event;
    if (!over) return;

    const id = active.id as string;
    const newStatus = over.id as SubmissionStatus;
    const sub = submissions.find((s) => s.id === id);
    if (!sub || sub.status === newStatus) return;

    // Intercept: moving to Lost
    if (newStatus === "lost") {
      setPendingMove({ id, from: sub.status, to: "lost" });
      return;
    }

    // Intercept: moving to Fully Complete from Finishing
    if (newStatus === "fully_complete" && sub.status === "finishing") {
      if (!sub.after_photos_uploaded || !sub.final_payment_confirmed) {
        setPendingMove({ id, from: sub.status, to: "fully_complete" });
        return;
      }
    }

    applyMove(id, newStatus);
  }

  const applyMove = useCallback(
    async (id: string, newStatus: SubmissionStatus, extra?: Record<string, unknown>) => {
      // Optimistic update
      setSubmissions((prev) =>
        prev.map((s) => (s.id === id ? { ...s, status: newStatus, ...extra } : s))
      );
      try {
        await fetch(`/api/admin/submissions/${id}`, {
          method: "PATCH",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ status: newStatus, ...extra }),
        });
      } catch {
        // On error, revert
        setSubmissions((prev) =>
          prev.map((s) =>
            s.id === id ? { ...s, status: (extra as { _from?: SubmissionStatus })?._from ?? s.status } : s
          )
        );
      }
      setPendingMove(null);
    },
    []
  );

  const pendingSub = pendingMove
    ? submissions.find((s) => s.id === pendingMove.id)
    : null;

  return (
    <>
      <DndContext sensors={sensors} onDragStart={handleDragStart} onDragEnd={handleDragEnd}>
        <div className="flex gap-3 overflow-x-auto pb-4 px-6" style={{ alignItems: "flex-start" }}>
          {KANBAN_COLUMNS.map((col) => (
            <KanbanColumn
              key={col}
              status={col}
              label={STATUS_LABELS[col]}
              submissions={grouped[col]}
              crewNames={crewNames}
              isLost={col === "lost"}
            />
          ))}
        </div>

        <DragOverlay>
          {draggingSubmission && (
            <div style={{ width: "160px" }}>
              <KanbanCard submission={draggingSubmission} crewNames={crewNames} />
            </div>
          )}
        </DragOverlay>
      </DndContext>

      {/* Lost modal */}
      <LostReasonModal
        open={pendingMove?.to === "lost"}
        onConfirm={(reason, note) => {
          if (pendingMove) applyMove(pendingMove.id, "lost", { lost_reason: reason, lost_note: note });
        }}
        onCancel={() => setPendingMove(null)}
      />

      {/* Finishing gate modal */}
      <FinishingGateModal
        open={pendingMove?.to === "fully_complete"}
        missingAfterPhotos={!pendingSub?.after_photos_uploaded}
        missingPayment={!pendingSub?.final_payment_confirmed}
        onConfirm={() => {
          if (pendingMove) applyMove(pendingMove.id, "fully_complete");
        }}
        onCancel={() => setPendingMove(null)}
      />
    </>
  );
}
```

```typescript
  doors: "doors-frames",
  door: "doors-frames",
  staircase: "staircases-railings",
  railing: "staircases-railings",
  kitchen: "kitchens",
  room: "rooms",
  lead: "lead-epa",
  epa: "lead-epa",
  furniture: "furniture",
  porch: "porches-exterior",
  exterior: "porches-exterior",
  service: "service-types",
  modifier: "modifiers-repair",
  repair: "modifiers-repair",
  admin: "admin-blocks",
  price: "price-benchmarks",
  benchmark: "price-benchmarks",
  glossary: "glossary",
};

const CATEGORY_LABELS: Record<string, string> = {
  "doors-frames": "Doors & Frames",
  "staircases-railings": "Staircases & Railings",
  kitchens: "Kitchens",
  rooms: "Rooms",
  "lead-epa": "Lead / EPA",
  furniture: "Furniture",
  "porches-exterior": "Porches & Exterior",
  "service-types": "Service Types",
  "modifiers-repair": "General Modifiers & Repair Tiers",
  "admin-blocks": "Admin Blocks",
  "price-benchmarks": "Price Benchmarks",
  glossary: "Architectural Glossary",
};

const ORDERED_KEYS = Object.keys(CATEGORY_LABELS);

function slugify(text: string): string {
  return text
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");
}

function detectCategory(heading: string): string | null {
  const lower = heading.toLowerCase();
  for (const [keyword, key] of Object.entries(CATEGORY_MAP)) {
    if (lower.includes(keyword)) return key;
  }
  return null;
}

const raw = fs.readFileSync(SOURCE, "utf-8");
const lines = raw.split("\n");

const categories: Record<string, CopyblockCategory> = {};
for (const key of ORDERED_KEYS) {
  categories[key] = { key, label: CATEGORY_LABELS[key], blocks: [] };
}

let currentCategoryKey: string | null = null;
let currentBlockTitle: string | null = null;
let currentBlockLines: string[] = [];

function flushBlock() {
  if (
    !currentCategoryKey ||
    !currentBlockTitle ||
    currentBlockLines.length === 0
  )
    return;
  const body = currentBlockLines.join("\n").trim();
  if (!body) return;
  const block: CopyblockItem = {
    key: slugify(currentBlockTitle),
    title: currentBlockTitle,
    body,
  };
  categories[currentCategoryKey].blocks.push(block);
}

for (const line of lines) {
  // H2 heading = new category
  if (line.startsWith("## ")) {
    flushBlock();
    currentBlockTitle = null;
    currentBlockLines = [];
    const heading = line.replace(/^##\s+/, "").trim();
    currentCategoryKey = detectCategory(heading);
    continue;
  }

  // H3 heading = new block within category
  if (line.startsWith("### ") && currentCategoryKey) {
    flushBlock();
    currentBlockLines = [];
    // Strip leading numbers/counts like "121 " or "(121) "
    currentBlockTitle = line
      .replace(/^###\s+/, "")
      .replace(/^\(?\d+\)?\s+/, "")
      .trim();
    continue;
  }

  // Accumulate body
  if (currentCategoryKey && currentBlockTitle) {
    currentBlockLines.push(line);
  }
}
flushBlock();

const library: CopyblockLibrary = ORDERED_KEYS.map(
  (key) => categories[key],
).filter((c) => c.blocks.length > 0);

fs.writeFileSync(OUTPUT, JSON.stringify(library, null, 2));
console.log(
  `Written ${library.reduce((n, c) => n + c.blocks.length, 0)} blocks across ${library.length} categories to ${OUTPUT}`,
);
```

```typescript
"use client";

import {
  DndContext,
  closestCenter,
  PointerSensor,
  KeyboardSensor,
  useSensor,
  useSensors,
  type DragEndEvent,
} from "@dnd-kit/core";
import {
  SortableContext,
  sortableKeyboardCoordinates,
  verticalListSortingStrategy,
  arrayMove,
} from "@dnd-kit/sortable";
import type { BidBlockDraft } from "@/lib/bid/types";
import { BlockInstance } from "./BlockInstance";

interface Props {
  blocks: BidBlockDraft[];
  signedPhotos: Array<{ path: string; url: string }>;
  isDraggingPhoto: boolean;
  onBlocksChange: (blocks: BidBlockDraft[]) => void;
  onTitleChange: (id: string, title: string) => void;
  onBodyChange: (id: string, body: string) => void;
  onRemove: (id: string) => void;
  onPhotoDrop: (blockId: string, photoPath: string) => void;
  onPhotoRemove: (blockId: string, photoPath: string) => void;
}

export function DocumentEditor({
  blocks, signedPhotos, isDraggingPhoto,
  onBlocksChange, onTitleChange, onBodyChange,
  onRemove, onPhotoDrop, onPhotoRemove,
}: Props) {
  const sensors = useSensors(
    useSensor(PointerSensor, { activationConstraint: { distance: 5 } }),
    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
  );

  function handleDragEnd(event: DragEndEvent) {
    const { active, over } = event;
    if (!over || active.id === over.id) return;

    const oldIndex = blocks.findIndex((b) => b.id === active.id);
    const newIndex = blocks.findIndex((b) => b.id === over.id);
    const reordered = arrayMove(blocks, oldIndex, newIndex).map((b, i) => ({
      ...b,
      sort_order: i,
    }));
    onBlocksChange(reordered);

    // Persist reordering
    reordered.forEach((b) => {
      fetch(`/api/admin/bid/blocks/${b.id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ sort_order: b.sort_order }),
      });
    });
  }

  if (blocks.length === 0) {
    return (
      <div className="border border-dashed border-[rgba(42,36,33,0.2)] rounded-sm p-10 text-center text-[#9e9087] text-sm font-sans">
        ← Click a block in the library to add it
      </div>
    );
  }

  return (
    <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={handleDragEnd}>
      <SortableContext items={blocks.map((b) => b.id)} strategy={verticalListSortingStrategy}>
        {blocks.map((block) => (
          <BlockInstance
            key={block.id}
            block={block}
            signedPhotos={signedPhotos}
            onTitleChange={onTitleChange}
            onBodyChange={onBodyChange}
            onRemove={onRemove}
            onPhotoDrop={onPhotoDrop}
            onPhotoRemove={onPhotoRemove}
            isDraggingPhoto={isDraggingPhoto}
          />
        ))}
      </SortableContext>
    </DndContext>
  );
}
```

```typescript
import { v4 as uuid } from "uuid";

}

export function BidBuilderClient({ submission, initialBlocks, signedPhotos, library }: Props) {
  const [blocks, setBlocks] = useState<BidBlockDraft[]>(
    initialBlocks.map((b, i) => ({
      id: b.id,
      source_key: b.source_key,
      title: b.title,
      body: b.body,
      photo_paths: b.photo_paths,
      sort_order: i,
    }))
  );
  const [amount, setAmount] = useState(submission.bid_amount?.toString() ?? "");
  const [startDate, setStartDate] = useState(submission.bid_start_date ?? "");
  const [draggingPhotoPath, setDraggingPhotoPath] = useState<string | null>(null);
  const saveTimers = useRef<Record<string, ReturnType<typeof setTimeout>>>({});

  const addedSourceKeys = new Set(blocks.map((b) => b.source_key));

  // Debounced save for title/body
  function scheduleBlockSave(id: string, field: "title" | "body", value: string) {
    if (saveTimers.current[id]) clearTimeout(saveTimers.current[id]);
    saveTimers.current[id] = setTimeout(() => {
      fetch(`/api/admin/bid/blocks/${id}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ [field]: value }),
      });
    }, 500);
  }

  const handleAddBlock = useCallback(async (item: CopyblockItem) => {
    const draft: BidBlockDraft = {
      id: uuid(),
      source_key: item.key,
      title: item.title,
      body: item.body,
      photo_paths: [],
      sort_order: blocks.length,
    };
    setBlocks((prev) => [...prev, draft]);
    const res = await fetch("/api/admin/bid/blocks", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        submission_id: submission.id,
        source_key: draft.source_key,
        title: draft.title,
        body: draft.body,
        sort_order: draft.sort_order,
      }),
    });
    if (res.ok) {
      const { block } = await res.json();
      // Replace temp uuid with DB id
      setBlocks((prev) => prev.map((b) => b.id === draft.id ? { ...b, id: block.id } : b));
    }
  }, [blocks.length, submission.id]);

  function handleTitleChange(id: string, title: string) {
    setBlocks((prev) => prev.map((b) => b.id === id ? { ...b, title } : b));
    scheduleBlockSave(id, "title", title);
  }

  function handleBodyChange(id: string, body: string) {
    setBlocks((prev) => prev.map((b) => b.id === id ? { ...b, body } : b));
    scheduleBlockSave(id, "body", body);
  }

  async function handleRemove(id: string) {
    setBlocks((prev) => prev.filter((b) => b.id !== id));
    await fetch(`/api/admin/bid/blocks/${id}`, { method: "DELETE" });
  }

  async function handlePhotoDrop(blockId: string, photoPath: string) {
    setBlocks((prev) =>
      prev.map((b) =>
        b.id === blockId && !b.photo_paths.includes(photoPath)
          ? { ...b, photo_paths: [...b.photo_paths, photoPath] }
          : b
      )
    );
    const block = blocks.find((b) => b.id === blockId);
```

```typescript
import { NextResponse } from "next/server";
import { renderToBuffer } from "@react-pdf/renderer";
import { createElement } from "react";
import { createServerSupabaseClient } from "@/lib/supabase/server";
import { createServiceSupabaseClient } from "@/lib/supabase/service";
import { resend } from "@/lib/email/resend";
import type { Submission, BidBlock } from "@/lib/types";
import { BidDocument } from "@/lib/bid/BidDocument";

const SITE_URL =
  process.env.NEXT_PUBLIC_SITE_URL ?? "https://suewheelerstl.com";
const FROM_EMAIL = process.env.FROM_EMAIL ?? "no-reply@suewheelerstl.com";
const CONTACT_EMAIL = process.env.CONTACT_EMAIL ?? "sue@suewheelerstl.com";

export async function POST(req: Request) {
  const supabaseAuth = await createServerSupabaseClient();
  const {
    data: { user },
  } = await supabaseAuth.auth.getUser();
  if (!user)
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

  const { submission_id, bid_amount, bid_start_date, send_to_client } =
    await req.json();
  if (!submission_id)
    return NextResponse.json(
      { error: "Missing submission_id" },
      { status: 400 },
    );

  const supabase = createServiceSupabaseClient();

  // Fetch submission and blocks
  const [{ data: submission }, { data: blocks }] = await Promise.all([
    supabase
      .from("submissions")
      .select()
      .eq("id", submission_id)
      .single<Submission>(),
    supabase
      .from("bid_blocks")
      .select()
      .eq("submission_id", submission_id)
      .order("sort_order")
      .returns<BidBlock[]>(),
  ]);

  if (!submission)
    return NextResponse.json(
      { error: "Submission not found" },
      { status: 404 },
    );
  if (!blocks)
    return NextResponse.json(
      { error: "Could not load blocks" },
      { status: 500 },
    );

  // Generate signed URLs for block photos
  const blocksWithPhotos = await Promise.all(
    blocks.map(async (block) => {
      const signedPhotoUrls: string[] = [];
      for (const path of block.photo_paths) {
        const { data } = await supabase.storage
          .from("submission-photos")
          .createSignedUrl(path, 60 * 60);
        if (data?.signedUrl) signedPhotoUrls.push(data.signedUrl);
      }
      return { ...block, signedPhotoUrls };
    }),
  );

  // Render PDF
  const docElement = createElement(BidDocument, {
    clientName: submission.name,
    address: submission.address,
    bidAmount: bid_amount ?? submission.bid_amount,
    bidStartDate: bid_start_date ?? submission.bid_start_date,
    blocks: blocksWithPhotos,
  });

  const pdfBuffer = await renderToBuffer(docElement);

  // Upload to Supabase Storage
  const timestamp = Date.now();
  const storagePath = `${submission_id}/estimate-${timestamp}.pdf`;
  const { error: uploadError } = await supabase.storage
    .from("bid-files")
    .upload(storagePath, pdfBuffer, {
      contentType: "application/pdf",
      upsert: false,
    });

  if (uploadError) {
    console.error("[generate-pdf] upload error", uploadError);
    return NextResponse.json({ error: "PDF upload failed" }, { status: 500 });
  }

  // Update submission with new bid_file_url and optional amount/start date
  const updatePayload: Record<string, unknown> = { bid_file_url: storagePath };
  if (bid_amount != null) updatePayload.bid_amount = bid_amount;
  if (bid_start_date) updatePayload.bid_start_date = bid_start_date;

  await supabase
    .from("submissions")
    .update(updatePayload)
    .eq("id", submission_id);

  // Generate a 30-min signed URL to return / for preview
  const { data: signedData } = await supabase.storage
    .from("bid-files")
    .createSignedUrl(storagePath, 60 * 30);

  const pdfUrl = signedData?.signedUrl ?? null;

  // Optionally send to client
  if (send_to_client && process.env.RESEND_API_KEY) {
    // Update bid status
    await supabase
      .from("submissions")
      .update({
        bid_status: "sent",
        bid_sent_at: new Date().toISOString(),
        bid_notes: blocks.map((b) => b.title).join(", "),
        status: "bid_sent",
      })
      .eq("id", submission_id);

    const portalUrl = `${SITE_URL}/my-request/${submission.client_token}`;
    const amount = bid_amount ?? submission.bid_amount;

    await resend.emails.send({
      from: FROM_EMAIL,
      to: [submission.email],
      replyTo: CONTACT_EMAIL,
      subject: `Your Estimate — Sue Wheeler Wood Refinishing`,
      html: `
        <p>Hi ${submission.name.split(" ")[0]},</p>
        <p>Your estimate${amount ? ` of <strong>$${amount.toLocaleString()}</strong>` : ""} is attached. You can also view it at your portal and accept online:</p>
        <p><a href="${portalUrl}">${portalUrl}</a></p>
        <p>Questions? Call Sue directly at (314) 367-6054.</p>
        <p style="color:#9e9087;font-size:12px;">Sue Wheeler Wood Refinishing · Est. 1989</p>
      `,
      attachments: [
        {
          filename: `estimate-${submission.name.replace(/\s+/g, "-")}.pdf`,
          content: Buffer.from(pdfBuffer).toString("base64"),
        },
      ],
    });
  }

  return NextResponse.json({ ok: true, pdfUrl });
}
```

```typescript
"use client";

import { useState } from "react";

export function ArchiveButton() {
  const [status, setStatus] = useState<"idle" | "running" | "done" | "error">("idle");
  const [count, setCount] = useState<number | null>(null);

  async function handleArchive() {
    if (!confirm("Archive all Fully Complete and Lost jobs older than 90 days?")) return;
    setStatus("running");
    const res = await fetch("/api/admin/archive", { method: "POST" });
    if (res.ok) {
      const { count: n } = await res.json();
      setCount(n);
      setStatus("done");
    } else {
      setStatus("error");
    }
  }

  return (
    <div>
      <button
        onClick={handleArchive}
        disabled={status === "running"}
        className="px-5 py-2.5 border border-[rgba(42,36,33,0.2)] text-xs font-sans font-semibold uppercase tracking-widest text-[#6B5E55] hover:border-[#2A2421] disabled:opacity-60 transition-colors"
        style={{ borderRadius: "2px" }}
      >
        {status === "running" ? "Archiving…" : "Archive Completed Jobs"}
      </button>
      {status === "done" && (
        <p className="text-xs font-sans text-[#2A7A2A] mt-2">
          ✓ {count} job{count !== 1 ? "s" : ""} archived.
        </p>
      )}
      {status === "error" && (
        <p className="text-xs font-sans text-[#A65D37] mt-2">Archive failed — try again.</p>
      )}
    </div>
  );
}
```
